pragma solidity 0.8.33;
contract Child {
    mapping(uint256 => uint256) s;
    function w(uint256 i) external { s[i] = 1; }
    function c(uint256 i) external { s[i] = 0; }
}
/// gasleft() semantics under EIP-8037 with and without a state-gas reservoir (real transactions).
contract Probe {
    event R(string k, int256 v);
    Child public ch;
    mapping(uint256 => uint256) s;
    constructor() { ch = new Child(); }
    function run(uint256 base) external {
        uint256 g0 = gasleft(); s[base] = 1; uint256 g1 = gasleft();
        emit R("own fresh SSTORE: gasleft delta", int256(g0) - int256(g1));
        g0 = gasleft(); ch.w(base); g1 = gasleft(); ch.c(base); uint256 g2 = gasleft();
        emit R("child 0->x: caller gasleft delta", int256(g0) - int256(g1));
        emit R("child x->0 restore: caller gasleft delta (neg = gasleft ROSE)", int256(g1) - int256(g2));
        (bool ok, ) = address(ch).call{gas: 30_000}(abi.encodeCall(Child.w, (base + 1)));
        emit R("child fresh SSTORE with {gas: 30000}: ok", ok ? int256(1) : int256(0));
        (ok, ) = address(ch).call{gas: 100_000}(abi.encodeCall(Child.w, (base + 2)));
        emit R("child fresh SSTORE with {gas: 100000}: ok", ok ? int256(1) : int256(0));
    }
}
