contract G7MutationCore is SuperPaymasterV55Test {
    function test_mutation_balance_settlement_noop() public {
        vm.mockCall(address(token), abi.encodeWithSignature("settleLocked(address,bytes32,uint256)"), abi.encode(uint256(0)));
        test_balance_mode_end_to_end();
    }
    function test_mutation_no_oog_old_guard() public {
        vm.prank(owner);
        (bool ok,) = address(sp).call(abi.encodeWithSignature("queueGasParams(uint32,uint32,uint32,uint32)", uint32(440000), uint32(160000), uint32(5000), uint32(340000)));
        require(ok);
        vm.warp(vm.getBlockTimestamp() + 48 hours + 1);
        vm.prank(owner);
        (ok,) = address(sp).call(abi.encodeWithSignature("executeGasParams()"));
        require(ok);
        sp.updatePrice();
        test_B1_no_oog_band_above_entry_guard();
    }
    function test_mutation_measured_postop_old_minimum() public {
        vm.prank(owner);
        (bool ok,) = address(sp).call(abi.encodeWithSignature("queueGasParams(uint32,uint32,uint32,uint32)", uint32(200000), uint32(160000), uint32(5000), uint32(175000)));
        require(ok);
        vm.warp(vm.getBlockTimestamp() + 48 hours + 1);
        vm.prank(owner);
        (ok,) = address(sp).call(abi.encodeWithSignature("executeGasParams()"));
        require(ok);
        sp.updatePrice();
        test_G_min_post_op_gas_covers_measured_postOp();
    }
    function test_mutation_floor_settlement_reverts() public {
        vm.mockCallRevert(address(token), abi.encodeWithSignature("settleLocked(address,bytes32,uint256)"), "mutation");
        test_TR1409_settles_at_MIN_POST_OP_GAS_worst_path();
    }
}
