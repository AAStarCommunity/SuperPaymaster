/// @dev Writes `n` fresh (zero -> non-zero) slots: one EIP-8037 SSTORE-set each.
contract R2aWriter {
    mapping(uint256 => uint256) public s;
    function w(uint256 from, uint256 n) external { for (uint256 i; i < n; i++) s[from + i] = 1; }
    function clear(uint256 from, uint256 n) external { for (uint256 i; i < n; i++) s[from + i] = 0; }
}

/// @dev Calls a target with an explicit gas limit and reports the gasleft() movement around it.
contract R2aCaller {
    function callWith(address t, uint256 g, bytes calldata d) external returns (bool ok, int256 delta) {
        uint256 g0 = gasleft();
        (ok, ) = t.call{gas: g}(d);
        delta = int256(g0) - int256(gasleft());
    }
}

// =====================================================================================
// Part A — gasleft() semantics probes (no SuperPaymaster)
// =====================================================================================
contract R2aGasleftProbe is Test {
    function _log(string memory k, int256 v) internal pure { console.log(string.concat("R2A|", k, "|", vm.toString(v))); }

    /// A1: the caller's gasleft() movement across (i) a child 0->x SSTORE, (ii) a child x->0 restore of
    ///     that same slot in the same transaction. Under EIP-8037 the restore refills state gas, which
    ///     revm credits back to `remaining` of the frames that spilled it: gasleft() can RISE across a call.
    function test_A1_gasleft_across_set_then_restore() public {
        R2aWriter wr = new R2aWriter();
        uint256 g0 = gasleft();
        wr.w(1000, 1);
        uint256 g1 = gasleft();
        wr.clear(1000, 1);
        uint256 g2 = gasleft();
        _log("A1.set_one_fresh_slot.caller_gas_consumed", int256(g0) - int256(g1));
        _log("A1.restore_same_slot.caller_gas_consumed(neg=gasleft_rose)", int256(g1) - int256(g2));
    }

    /// A2: the restore in a frame that did NOT spill (its own spill counter is 0): the refill lands in
    ///     that frame's reservoir and only reaches `remaining` of an ancestor that spilled.
    function test_A2_restore_from_non_spilling_intermediate() public {
        R2aWriter wr = new R2aWriter();
        R2aCaller c = new R2aCaller();
        wr.w(2000, 3);
        (bool ok, int256 d) = c.callWith(address(wr), 500_000, abi.encodeCall(R2aWriter.clear, (2000, 3)));
        require(ok, "clear");
        _log("A2.intermediate_frame_delta_restoring_3_slots(neg=gasleft_rose)", d);
    }

    /// A3: a child given an explicit gas limit tries one fresh SSTORE. With no reservoir (forge default,
    ///     tx gas cap disabled) the state gas must fit the child's own limit; with a reservoir
    ///     (--enable-tx-gas-limit, tx gas limit 2^30 > 2^24) the state gas comes from the reservoir.
    function test_A3_state_gas_vs_call_gas_limit() public {
        R2aWriter wr = new R2aWriter();
        R2aCaller c = new R2aCaller();
        uint32[6] memory gs = [uint32(25_000), 30_000, 50_000, 100_000, 125_000, 150_000];
        for (uint256 i; i < gs.length; i++) {
            (bool ok, int256 d) = c.callWith(address(wr), gs[i], abi.encodeCall(R2aWriter.w, (3000 + i * 10, 1)));
            _log(string.concat("A3.one_fresh_sstore_with_call_gas_", vm.toString(uint256(gs[i])), ".ok"), ok ? int256(1) : int256(0));
            _log(string.concat("A3.one_fresh_sstore_with_call_gas_", vm.toString(uint256(gs[i])), ".caller_delta"), d);
        }
    }
}

contract R2aEP is SuperPaymasterV55GasTest {
    bytes32 constant POSTOP_REVERT_TOPIC = keccak256("PostOpRevertReason(bytes32,address,uint256,bytes)");
    bytes32 constant PREFUND_LOW_TOPIC = keccak256("UserOperationPrefundTooLow(bytes32,address,uint256)");
    bytes32 constant USER_REVERT_TOPIC = keccak256("UserOperationRevertReason(bytes32,address,uint256,bytes)");

    uint8 constant BAL = 1;
    uint8 constant CRED = 2;

    function _creditUser(uint256 salt) internal returns (address u) {
        u = _account(salt);
        vm.prank(address(registry));
        sp.updateSBTStatus(u, true);
        registry.setCreditLimit(u, 50_000 ether);

    function _balanceUser(uint256 salt) internal returns (address u) {
        u = _account(salt);
        vm.prank(address(registry));
        sp.updateSBTStatus(u, true);
        vm.prank(operator);
        IV2Ext(address(token)).mint(u, 10_000 ether);
    }

    function _mk(address u, uint256 nonce, bytes memory callData, Lim memory l) internal view returns (PackedUserOperation memory op) {
        op.sender = u;
        op.nonce = nonce;
        op.callData = callData;
        op.accountGasLimits = bytes32(abi.encodePacked(l.ver, l.callG));
        op.preVerificationGas = l.pvg;
        op.gasFees = bytes32(abi.encodePacked(uint128(1 gwei), uint128(1 gwei)));
        op.paymasterAndData = abi.encodePacked(address(sp), l.pmVer, l.postOp, operator, type(uint256).max, address(token), uint8(0));
        _sign(op);
    }

    function _a0(PackedUserOperation memory op) internal returns (uint256 a0) {
        uint256 snap = vm.snapshot();
        bytes32 h = entryPoint.getUserOpHash(op);
        vm.prank(EP);
        try sp.validatePaymasterUserOp(op, h, _prefund(op)) returns (bytes memory ctx, uint256) {
            if (ctx.length >= 352) a0 = abi.decode(ctx, (SuperPaymaster.OpCtx)).a0;
        } catch {}
        vm.revertTo(snap);
    }

    function _prefund(PackedUserOperation memory op) internal pure returns (uint256) {
        uint256 ver = uint128(uint256(op.accountGasLimits) >> 128);
        uint256 cg = uint128(uint256(op.accountGasLimits));
        bytes memory p = op.paymasterAndData;
        uint256 pmv; uint256 pmp;
        assembly { pmv := shr(128, mload(add(p, 52))) pmp := shr(128, mload(add(p, 68))) }
        return (ver + cg + pmv + pmp + op.preVerificationGas) * 1 gwei;
    }

    function _exec(PackedUserOperation memory op) internal returns (Out memory o) {
        address u = op.sender;
        o.a0 = _a0(op);
        uint256 rev0 = sp.protocolRevenue();
        uint256 dep0 = entryPoint.balanceOf(address(sp));
        uint256 bal0 = token.balanceOf(u);
        uint256 debt0 = token.debts(u);
        PackedUserOperation[] memory ops = new PackedUserOperation[](1);
        ops[0] = op;
        vm.recordLogs();
        uint256 g0 = gasleft();
        try entryPoint.handleOps(ops, payable(beneficiary)) {} catch { o.bundleReverted = true; }
        o.frameGas = g0 - gasleft();
        Vm.Log[] memory logs = vm.getRecordedLogs();
        for (uint256 i; i < logs.length; i++) {
            if (logs[i].emitter != EP || logs[i].topics.length == 0) continue;
            bytes32 t = logs[i].topics[0];
            if (t == USEROP_EVENT) {
                (, bool s, uint256 c, uint256 g) = abi.decode(logs[i].data, (uint256, bool, uint256, uint256));
                o.success = s; o.cost = c; o.used = g;
            } else if (t == POSTOP_REVERT_TOPIC) {
                o.postOpRev = true;
                (, bytes memory r) = abi.decode(logs[i].data, (uint256, bytes));
                // EntryPoint wraps the paymaster's revert data: PostOpReverted(bytes inner). Unwrap it.
                if (r.length >= 68 && bytes4(r) == bytes4(keccak256("PostOpReverted(bytes)"))) {
                    bytes memory tail = new bytes(r.length - 4);
                    for (uint256 j; j < tail.length; j++) tail[j] = r[j + 4];
                    r = abi.decode(tail, (bytes));
                }
                o.postOpReasonLen = r.length; // inner paymaster revert data; 0 = out of gas / empty revert
                if (r.length >= 4) o.postOpSel = bytes4(r);
            } else if (t == PREFUND_LOW_TOPIC) {
                o.prefundLow = true;
            } else if (t == USER_REVERT_TOPIC) {
                o.userRevert = true;
            }
        }
        if (o.bundleReverted) return o;
        o.charge = sp.protocolRevenue() - rev0;
        o.depDelta = dep0 - entryPoint.balanceOf(address(sp));
        o.need = _fullCostCharge(o.cost);
        uint256 bal1 = token.balanceOf(u);
        o.userBurn = bal0 > bal1 ? bal0 - bal1 : 0;
        o.debtDelta = token.debts(u) - debt0;
    }

    function _report(string memory name, Out memory o) internal pure {
        _log(_k(name, "bundleReverted"), o.bundleReverted ? 1 : 0);
        _log(_k(name, "success"), o.success ? 1 : 0);
        _log(_k(name, "userRevertEvent"), o.userRevert ? 1 : 0);
        _log(_k(name, "postOpReverted"), o.postOpRev ? 1 : 0);
        _log(_k(name, "postOpRevertSelector"), uint256(uint32(o.postOpSel)));
        _log(_k(name, "postOpRevertReasonLen"), o.postOpReasonLen);
        _log(_k(name, "prefundTooLow"), o.prefundLow ? 1 : 0);
        _log(_k(name, "actualGasUsed"), o.used);
        _log(_k(name, "actualGasCost_wei"), o.cost);
        _log(_k(name, "spEpDepositDelta_wei"), o.depDelta);
        _log(_k(name, "charge_aPNTs"), o.charge);
        _log(_k(name, "need_aPNTs(full cost at snapshot)"), o.need);
        _log(_k(name, "chargeCoversNeed(I9/G2)"), o.charge >= o.need ? 1 : 0);
        _log(_k(name, "a0(clamp)"), o.a0);
        _log(_k(name, "chargeClampedAtA0"), (o.a0 != 0 && o.charge == o.a0) ? 1 : 0);
        _logi(_k(name, "charge_minus_need_aPNTs"), int256(o.charge) - int256(o.need));
        _log(_k(name, "userBurn_xPNTs"), o.userBurn);
        _log(_k(name, "userDebtDelta_xPNTs"), o.debtDelta);
        _log(_k(name, "testFrameGas_handleOps"), o.frameGas);
    }

    // -------------------------------------------------------------------------------
    // B2 — postOp segment that EntryPoint charges (W_postop at EntryPoint level):
    //      F(real postOp) - F(postOp mocked to return immediately) = consumed - 10% penalty

    function _postOpSegment(string memory name, address u, bytes memory callData, Lim memory l) internal {
        uint256 snap = vm.snapshot();
        Out memory real = _exec(_mk(u, 0, callData, l));
        vm.revertTo(snap);
        snap = vm.snapshot();
        vm.mockCall(address(sp), abi.encodeWithSelector(IPaymaster.postOp.selector), "");
        Out memory mocked = _exec(_mk(u, 0, callData, l));
        vm.clearMockedCalls();
        vm.revertTo(snap);
        _log(_k(name, "F_real"), real.used);
        _log(_k(name, "F_postOpMocked"), mocked.used);
        _logi(_k(name, "W_postop_EP(=(F_real-F_mocked)*10/9)"), (int256(real.used) - int256(mocked.used)) * 10 / 9);
        _log(_k(name, "real.success"), real.success ? 1 : 0);
        _log(_k(name, "real.postOpReverted"), real.postOpRev ? 1 : 0);
        _log(_k(name, "real.prefundTooLow"), real.prefundLow ? 1 : 0);
    }

    // -------------------------------------------------------------------------------
    // B3 — smallest limits through EntryPoint (binary search; success predicate per limit)
    // -------------------------------------------------------------------------------
    function _okPmVer(address u, bytes memory cd, uint128 g) internal returns (bool) {
        uint256 snap = vm.snapshot();
        Lim memory l = _lim(); l.pmVer = g;
        Out memory o = _exec(_mk(u, 0, cd, l));
        vm.revertTo(snap);
        return !o.bundleReverted;
    }

    function _okVer(address u, bytes memory cd, uint128 g, bytes memory initCode) internal returns (bool) {
        uint256 snap = vm.snapshot();
        Lim memory l = _lim(); l.ver = g;
        PackedUserOperation memory op = _mk(u, 0, cd, l);
        if (initCode.length > 0) { op.initCode = initCode; _sign(op); }
        Out memory o = _exec(op);
        vm.revertTo(snap);
        return !o.bundleReverted;
    }

    function _okCall(address u, bytes memory cd, uint128 g) internal returns (bool) {
        uint256 snap = vm.snapshot();
        Lim memory l = _lim(); l.callG = g;
        Out memory o = _exec(_mk(u, 0, cd, l));
        vm.revertTo(snap);
        return !o.bundleReverted && o.success && !o.postOpRev;
    }

    function _okPostOp(address u, bytes memory cd, uint128 g) internal returns (bool settled, bytes4 sel, uint256 len) {
        uint256 snap = vm.snapshot();
        Lim memory l = _lim(); l.postOp = g;
        Out memory o = _exec(_mk(u, 0, cd, l));
        vm.revertTo(snap);
        settled = !o.bundleReverted && !o.postOpRev && !o.prefundLow;
        sel = o.postOpSel; len = o.postOpReasonLen;
    }

            uint128 lo = 175_000; uint128 hi = 3_000_000;
            (bool ok, , ) = _okPostOp(u, cd, hi);
            require(ok, "hi must settle");
            (ok, , ) = _okPostOp(u, cd, lo);
            if (ok) { _log(string.concat("B3.min_postOpGasLimit_EP.", name), lo); continue; }
            while (hi - lo > 1) { uint128 mid = (lo + hi) / 2; (ok, , ) = _okPostOp(u, cd, mid); if (ok) hi = mid; else lo = mid; }

            _log(string.concat("B3.min_postOpGasLimit_EP.", name, ".failure_at_min-1.reasonLen"), len);

    }

    // -------------------------------------------------------------------------------
    // B4 — no-OOG band (direct postOp call from the EntryPoint address, as EntryPoint calls it)
    // -------------------------------------------------------------------------------
    function _ctx(uint8 mode, uint256 salt, uint128 postOpGas) internal returns (bytes memory ctx, bytes32 h) {
        address u = mode == CRED ? _creditUser(salt) : _balanceUser(salt);
        Lim memory l = _lim(); l.postOp = postOpGas;
        PackedUserOperation memory op = _mk(u, 0, "", l);
        h = entryPoint.getUserOpHash(op);
        vm.prank(EP);
        uint256 vd;
        (ctx, vd) = sp.validatePaymasterUserOp(op, h, 1e16);
        require(vd & 1 == 0, "validation failed");
        require(abi.decode(ctx, (SuperPaymaster.OpCtx)).mode == mode, "mode");
    }

    /// 0 = settled, 1 = PostOpGasTooLow (guard), 2 = any other failure (OOG band)
    function _directPostOp(bytes memory ctx, uint256 g) internal returns (uint8) {
        bytes memory cd = abi.encodeCall(IPaymaster.postOp, (IPaymaster.PostOpMode.opSucceeded, ctx, 1e14, 1 gwei));
        uint256 snap = vm.snapshot();
        vm.prank(EP);
        (bool s, bytes memory ret) = address(sp).call{gas: g}(cd);
        vm.revertTo(snap);
        if (s) return 0;
        if (keccak256(ret) == keccak256(abi.encodeWithSelector(SuperPaymasterStorage.PostOpGasTooLow.selector))) return 1;
        return 2;
    }

    function _bandFor(uint8 mode, string memory name, uint32 settle) internal returns (uint256 gOk, uint256 gGuard, uint256 bandFails) {
        uint256 snap = vm.snapshot();
        _setGP(settle + 20_000, settle, 5_000, 175_000);
        (bytes memory ctx, ) = _ctx(mode, 120 + mode + settle / 1000, settle + 20_000);
        // smallest settling call gas
        uint256 lo = 50_000; uint256 hi = 3_000_000;
        while (hi - lo > 1) { uint256 mid = (lo + hi) / 2; if (_directPostOp(ctx, mid) == 0) hi = mid; else lo = mid; }
        gOk = hi;
        // smallest call gas that passes the entry guard
        lo = 50_000; hi = 3_000_000;
        while (hi - lo > 1) { uint256 mid = (lo + hi) / 2; if (_directPostOp(ctx, mid) != 1) hi = mid; else lo = mid; }
        gGuard = hi;
        for (uint256 g = gGuard; g < gOk; g += 1_000) if (_directPostOp(ctx, g) == 2) bandFails++;
        string memory p = string.concat("B4.", name, ".SETTLE_", vm.toString(uint256(settle)));
        _log(_k(p, "min_settling_call_gas"), gOk);
        _log(_k(p, "min_call_gas_passing_guard"), gGuard);
        _log(_k(p, "oog_band_width(gOk-gGuard,0=none)"), gOk > gGuard ? gOk - gGuard : 0);
        _log(_k(p, "oog_band_failures_step1k"), bandFails);
        vm.revertTo(snap);
    }

    function test_B4_no_oog_band() public {
        _setUpAll();

        }
    }

    // -------------------------------------------------------------------------------
    // R2-F2 addition (not in the R-AMS-2a scratch): per-scenario minimum limits on the axes of the B-layer
    // matrix (fresh vs warm SP, first vs subsequent op of a sender, SP_RENEW, balance vs credit, initCode),
    // GasParams = the (c)/(d) tuple 440k/420k/5k/340k, success predicate = the B-layer probe's (handleOps
    // does not revert, UserOperationEvent.success, no PostOpRevertReason, no prefund-too-low). Run with --isolate.
    // -------------------------------------------------------------------------------
    function _mkF(address u, uint256 nonce, Lim memory l, uint8 flags, bytes memory initCode) internal view returns (PackedUserOperation memory op) {
        op = _mk(u, nonce, "", l);
        op.paymasterAndData = abi.encodePacked(address(sp), l.pmVer, l.postOp, operator, type(uint256).max, address(token), flags);
        if (initCode.length > 0) op.initCode = initCode;
        _sign(op);
    }

    /// which: 0 = verificationGasLimit, 1 = callGasLimit, 2 = paymasterVerificationGasLimit, 3 = paymasterPostOpGasLimit
    function _okF(address u, uint256 nonce, uint8 flags, bytes memory initCode, uint8 which, uint128 g) internal returns (bool) {
        uint256 snap = vm.snapshot();
        Lim memory l = _lim();
        if (initCode.length > 0) l.ver = 5_000_000;
        if (which == 0) l.ver = g; else if (which == 1) l.callG = g; else if (which == 2) l.pmVer = g; else l.postOp = g;
        Out memory o = _exec(_mkF(u, nonce, l, flags, initCode));
        vm.revertTo(snap);
        return !o.bundleReverted && o.success && !o.postOpRev && !o.prefundLow;
    }

    function _minsF(string memory name, address u, uint256 nonce, uint8 flags, bytes memory initCode) internal {
        string[4] memory f = ["v", "call", "pmV", "pmP"];
        uint128[4] memory los = [uint128(1_000), 1_000, 10_000, 440_000];
        for (uint8 w; w < 4; w++) {
            uint128 lo = los[w]; uint128 hi = initCode.length > 0 && w == 0 ? 20_000_000 : 5_000_000;
            if (!_okF(u, nonce, flags, initCode, w, hi)) { _log(string.concat("F2.", name, ".", f[w], ".fails_at_hi"), hi); continue; }
            if (_okF(u, nonce, flags, initCode, w, lo)) { _log(string.concat("F2.", name, ".min_", f[w], "(=lower bound)"), lo); continue; }
            while (hi - lo > 1) { uint128 mid = (lo + hi) / 2; if (_okF(u, nonce, flags, initCode, w, mid)) hi = mid; else lo = mid; }
            _log(string.concat("F2.", name, ".min_", f[w]), hi);
        }
    }
